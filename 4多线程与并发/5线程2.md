# POSIX单个信号量

分为：有名信号量和无名信号量

- **有名信号量**：
  - 通过一个名字（通常是文件系统路径）来标识，可以用于**不相关进程**之间的同步。
  - 可以用于**进程或线程间 同步/互斥**，在文件系统中有一个入口（有一个文件/`inode`），信号量的对象（值）存在于内核中。即可以用于人已经进程间，也可用于线程间同步。
- **无名信号量**（基于内存的信号量）：
  - 没有名字，
  - 存在于内存中（通常是在共享内存区），
  - 如果这段内存在一个 **内核的共享内存中**，进程可以访问，线程也可以访问  -----》既可以用于进程间的同步/互斥，也可以用于线程间同步/互斥。
  - 如果这段内容在一个 **进程的地址空间**，此时，只用于线程间(线程内部) 同步/互斥
  - 主要用于**线程之间**或**相关进程**（如父子进程）之间的同步。

用`sem_t`来描述`POSIX`信号量

## POSIX信号量的P/V操作

#### P操作 sem_wait

````c++
#include<semaphore.h>

int sem_wait(sem_t *sem) 	// P操作（等待/获取信号量）
/*
	作用：
		死等上锁
	@sem:
		指向一个POSIX信号量的空间地址
		需要上锁的那个信号量
	@return：
		成功返回0，表示获取该信号量成功
		失败返回-1，表示该信号获取失败
*/
````

- 这是最常用的函数。它尝试将信号量的值减1。
- 如果信号量值大于0，则立即减1并返回。
- 如果信号量值为0，则线程/进程会**阻塞**，直到信号量值变为大于0（即有其他进程/线程执行了 `sem_post`）。

```c++
int sem_trywait(sem_t *sem)	 //非阻塞的 P操作
/*
	作用：
		尝试上锁，能够上锁就上锁，不能上锁就返回
	@sem:
		指向一个POSIX信号量的空间地址
		需要上锁的那个信号量
	@return:
		成功返回0，表示获取该信号量成功
		失败返回-1，表示该信号获取失败
*/
```

- 如果信号量立即可用（值>0），则减1并返回0。
- 如果信号量不可用（值=0），**不会阻塞**，而是立即返回-1，并设置 `errno` 为 `EAGAIN`。

```c++
int sem_timedwait(sem_t *sem,const struct timespec *abs_timeout)
/*	
	作用：
		限时上锁，超过绝对时间就返回
	@sem:
		指向一个POSIX信号量的空间地址
		需要上锁的那个信号量
	@abs_timeout:
		绝对时间：绝对时间 = 当前时间+相对时间
	@return:
		成功返回0，表示获取该信号量成功
		失败返回-1，表示该信号获取失败
	如：
	struct timespec ts;
	clock_gettime(CLOCK_REALTIME,&ts);//获取绝对时间
	//等待5s,30ms
	ts.tv_sec += 5;
	ts.tv_nsec += 30000000;//纳秒
	if(ts.tv_nsec >= 1000000000)
	{
		ts.tv_sec++;
		ts.tv_nsec -= 1000000000;
	}
	int r = sem_timedwait(sem,&ts);
*/
```

#### V操作 sem_post

```c++
#include<semaphore.h>

int sem_post(set_t *sem);
/*
	作用：
		解锁
	@sem:
		需要解锁的信号量指针

*/
```

- 将信号量的值**加1**。
- 如果有其他进程/线程正在因等待这个信号量而阻塞，此操作会唤醒其中一个。

# 示例：POSIX

**POSIX_SEM_A.cpp**

```c++
#include<iostream>
#include<semaphore.h>
#include<fcntl.h>
#include<sys/stat.h>

#define SEMNAME "/sem"
int main()
{   
    //创建一个有名信号量
    sem_t *sem_id = sem_open(SEMNAME,O_CREAT,0777,1);

    //p操作 上锁
    sem_wait(sem_id);
    std::cout<<"已上锁"<<std::endl;
    FILE *file = fopen("1.txt","a+");
    std::cout<<"请输入要写入的内容："<<std::endl;
    std::string data;
    std::cin>>data;
    fwrite(data.c_str(),1,data.size(),file);
    fclose(file);
    //v操作 解锁
    sem_post(sem_id);
    std::cout<<"已解锁"<<std::endl;

    //关闭信号量
    sem_close(sem_id);

    //销毁信号量
    sem_unlink(SEMNAME);
    return 0;
}
```

# 线程互斥锁 `pthread_mutex_t`

线程互斥锁也是信号量，只不过线程互斥锁，存在于进程地址空间，用于线程间同步和互斥操作，线程 互斥锁它的效率相对信号量来说要高。

**线程互斥锁**：使用 `pthread_mutex_t` 的类型来描述一个锁

## 初始化线程互斥锁 `pthread_mutex_init`

```c++

pthread_mutexattr_t *restrict attr);
/*
    @描述：
    	初始一个线程互斥锁
    @mutex:
   	 	需要初始化的线程互斥锁地址
   	 	
        pthread_mutex_t mutex; // 创建了一个互斥锁
        &mutex ---> 互斥锁地址
        
    @attr:
        线程互斥锁的属性，一般为NULL，采用默认属性

        	线程的互斥锁默认设置 1 unlock

```

## 线程互斥锁的`PV`操作

### `P` 操作（上锁操作）`lock` `trylock `  `timedlock`

````c++
int pthread_mutex_lock(pthread_mutex_t *mutex);
/*
    作用：
    	死等上锁，如果该锁没有被释放，则会一直阻塞在此函数，等待该锁被释放。
    @mutex：
    	需要上锁的互斥锁指针
    @return：	
        成功返回0，表示获取到了该互斥锁
        返回-1，表示获取出错，没有获取到互斥锁
*/
int pthread_mutex_trylock(pthread_mutex_t *mutex);
/*
    作用：
    	尝试上锁，尝试性上锁，如果该锁没被释放，那么立即返回执行后面的代码。
    @mutex：
    	需要上锁的互斥锁指针
    @return：
        成功返回0，表示获取到了该互斥锁
        其他值，表示没有获取到互斥锁
*/
int pthread_mutex_timedlock(pthread_mutex_t *mutex,struct timespec
*abs_timeout);
/*
    作用：
        限时上锁，如果该锁没有被释放，则会一直阻塞在此函数中一段实际，等待该锁被释放。如果时间
        过了还没有被释放，那么果断放弃执行后面的代码。
    @mutex:
    	需要上锁的互斥锁指针
    @abs_timeout:
    	绝对时间（超时时间），上锁的时间范围。
    @return：
        成功返回0，表示获取到了该互斥锁
        其他值，表示没有获取到互斥锁
*/
````

## 示例1：互斥锁 实现文档内容复制

 `pthread_mutex_1.cpp` 

```c++
#include<iostream>
#include<pthread.h>
#include<unistd.h>

//创建一个互斥锁
pthread_mutex_t mutex;

//偏移量数据作为共享资源
int seek_off = 0;

void * myThread(void *args)
{
    //
    FILE *file_ptr1 = fopen("1.txt","r");
    FILE *file_ptr2 = fopen("2.txt","a+");

    //循环读取文件内容，然后写入到另一个文件中
    while(!feof(file_ptr1))//feof返回值为0表示未读到文件末尾
    {
        pthread_mutex_lock(&mutex);//死等上锁
        sleep(1);
        char buff[1024] = {0};

        //定位光标位置
        fseek(file_ptr1,seek_off,SEEK_SET);

        int offset = fread(buff,1,1024,file_ptr1);//fread读取成功返回读取到的数量
        if(!offset)
        {
            pthread_mutex_unlock(&mutex);//解锁
            break;
        }

        seek_off += offset;//同步偏移量
        
        std::cout << buff << std::endl;
        fwrite(buff,1,std::string(buff).size(),file_ptr2);
        pthread_mutex_unlock(&mutex);//解锁
    }

    // 关闭文件
    fclose(file_ptr1);
    fclose(file_ptr2);

    return nullptr;
}
int main()
{   
    //定义线程ID号集
    pthread_t tid[5];

    //初始化线程互斥锁
    pthread_mutex_init(&mutex,NULL);

    //通过循环创建多个线程
    for(int i = 0;i<5;i++)
    {
        int pthread_create_i = pthread_create(tid+i,NULL,myThread,nullptr);
        if(pthread_create_i)
        {
            perror("创建线程失败");
        }
    }

    //回收线程资源
    for(int i = 0 ;i<5;i++)
    {
        int pthread_join_i = pthread_join(tid[i],nullptr);
        if(pthread_join_i)
        {
            perror("线程回收失败");
        }
    }

    //销毁线程互斥锁
    pthread_mutex_destroy(&mutex);
    
    return 0;
}
```

## 示例2：互斥锁 实现登录等待

`pthread_mutex_等待应用场景.cpp`

```c++
#include <iostream>
#include <pthread.h>
#include <unistd.h>
#include <time.h>
#include <stdlib.h>

pthread_mutex_t mutex;//创建互斥锁

bool state = false;// 登录状态标志

void *connectServer(void *args)//连接服务器的线程函数
{
    pthread_mutex_lock(&mutex);//获取互斥锁
    for(int i = 0;i < 5000;i++){
        usleep(2000);//循环5000次，每次休眠2000微秒
    }
    state = true;// 在互斥锁保护范围内修改state
    pthread_mutex_unlock(&mutex);
    
    return nullptr;
}

void *signWindow(void *args)//显示登录窗口的线程函数
{
    int dotCount = 0;
    while(1)
    {
        pthread_mutex_trylock(&mutex);
        if(state)
        {
            std::cout << std::endl << "登录成功" << std::endl;
            pthread_mutex_unlock(&mutex);
            break;
        }
        pthread_mutex_unlock(&mutex);

        system("clear");
        std::cout << "登录中";

        for(int i = 0;i < dotCount;i++)
        {
            std::cout << ".";
        }   
        std::cout << std::flush;  // 刷新输出缓冲区
        dotCount = (dotCount + 1) % 4;  // 点数量在0-3之间循环
        usleep(500000);  // 500ms延迟，使动画可见

        //第1次while循环：dotCount = 0 → 显示"登录中",休眠500ms
        //第2次while循环：dotCount = 1 → 显示"登录中.",休眠500ms
        //第3次while循环：dotCount = 2 → 显示"登录中..",休眠500ms
        //第4次while循环：dotCount = 3 → 显示"登录中...",休眠500ms
        //第5次while循环：dotCount = 0 → 回到"登录中",休眠500ms
        //第6次while循环：dotCount = 1 → 回到"登录中.",休眠500ms
        //如此循环...
    }
    return nullptr;
}


int main()
{
    pthread_mutex_init(&mutex,nullptr);//初始化一个互斥锁
    //创建连接服务器的线程
    pthread_t tid_connect;
    pthread_create(&tid_connect,nullptr,connectServer,nullptr);
    //创建显示登录窗口的线程
    pthread_t tid_sign;
    pthread_create(&tid_sign,nullptr,signWindow,nullptr);

    pthread_join(tid_connect,nullptr);
    pthread_join(tid_sign,nullptr);

    pthread_mutex_destroy(&mutex);//销毁互斥锁
    return 0;
}
```

### 程序功能

这是一个模拟登录过程的C++多线程程序，包含两个并行执行的线程：

1. **`connectServer`线程**：模拟连接服务器的过程
2. **`signWindow`线程**：显示登录界面的动画效果

1） **全局变量和互斥锁**

- `mutex`：用于线程间同步的互斥锁
- `state`：表示登录是否成功的标志，初始为`false`

2）**主线程 `main`  初始化互斥锁，创建两个线程，并等待它们完成，然后销毁互斥锁。**

3）**连接服务器线程（`connectServer`）：**

- **功能**：模拟连接服务器的过程

  - 获取互斥锁
    - 循环5000次，每次休眠2000微秒（5000次 x 2000微秒 = 10,000,000微秒；总共10秒），模拟连接服务器过程中的一些操作

  - 在锁保护下将`state`设置为`true`（表示登录成功）

  - 释放互斥锁

4） **登录窗口线程 (`signWindow`)**：

- **功能**：显示登录界面和动画效果

  - 使用`dotCount`变量控制动态点的数量

  - 使用`pthread_mutex_trylock()`尝试获取锁，不会阻塞线程

  - 检查登录状态，如果成功则显示"登录成功"并退出

  - 显示动态的"登录中..."动画，点数量在0-3之间循环变化
    - 每次循环后延迟500毫秒，使动画效果可见

### 程序执行流程

1. 主线程初始化互斥锁并创建两个线程
2. 两个线程开始并行执行：
   - 连接线程：获取锁 → 等待10秒 → 设置state=true → 释放锁
   - 界面线程：循环检查state → 显示动画 → 延迟 → 再次检查
3. 当界面线程检测到state为true时，显示"登录成功"并退出
4. 主线程等待两个线程结束后，销毁互斥锁并退出程序

### 关键特点

1. **线程同步**：使用互斥锁保护共享变量`state`
2. **非阻塞检查**：界面线程使用`trylock`避免阻塞动画显示
3. **动态动画**：通过循环计数实现"登录中"、"登录中."、"登录中.."、"登录中..."的动画效果
4. **时间控制**：连接过程固定10秒，动画刷新间隔500毫秒

**生产者消费者**模型：利用厂商和消费者关系，由生产者线程进行生产（**产生任务**），再由消费者线程消费（**执行任务**），没有任务的时候，消费者等待生产者产生任务。

当缓冲区（**生产者没有产出的时候**）没有数据的时候，（**消费者**）应该怎么办？

- 不停的去测试，看有没有数据。 
  - 轮询访问，但是轮询有缺陷：一直在访问，浪费CPU资源。轮询有时间差，占用**总线**： `Is always busy`
- 让出`CPU`,当有数据的时候，再唤醒我（ `wake up` ）,**线程条件变量**： `同步`

## 线程条件变量 `pthread_cond_t`

**线程条件变量**：在多线程程序设计中，可以用 `条件变量` 为表示一个特定的**条件**或者是**事件**

至于条件变量，到底是一个什么事件或者说表示一个什么条件？完全由**程序猿**去解释这个条件变量所代 表的含义。

`pthread_cond_t` ：来描述一个条件变量（类型）

在条件变量上的三种操作： 

- 初始化 
- 等待一个条件变量（**等待该条件变量所表示的事件**） 
- 唤醒一个线程/触发条件变量（**唤醒了正在等待该事件的线程**）

```c++
int data = 0;

main : 主线程：生产者：包工头
	data = 1;//消息，通知，事件

t1：子线程：消费者：牛马
    when data == 1;
    	干活
    data = 0;
```

## 线程条件变量`API`

### 初始化/销毁条件变量  `pthread_cond_destroy`, `pthread_cond_init` 

```c++
NAME
    pthread_cond_destroy, pthread_cond_init 
    — destroy and initialize condition variables     (销毁和初始化 条件变量)
    
SYNOPSIS
    
    #include <pthread.h>
    int pthread_cond_destroy(pthread_cond_t *cond);
    /*
        @描述：
        	销毁一个条件变量
        @cond：
        	需要销毁条件变量指针
    */
	int pthread_cond_init(pthread_cond_t *restrict cond,const pthread_condattr_t *restrict attr);
    /*
        @描述：
        	初始化一个条件变量
        @cond：
        	需要初始化的条件变量的指针
        @attr：
        	初始化的条件变量的属性，一般为NULL，采用默认设置
        @return：
        	成功返回0，失败返回其他值
    */

```

### 等待一个条件变量 `pthread_cond_timedwait`, `pthread_cond_wait`

```c++
NAME
	pthread_cond_timedwait, pthread_cond_wait
    — wait on a condition(在一个条件下等待)
    
SYNOPSIS   
    
    #include <pthread.h>
    int pthread_cond_timedwait(pthread_cond_t *restrict cond,pthread_mutex_t *restrict mutex,
    const struct timespec *restrict abstime);
    /*
        @描述：
            限时等待条件变量
        @cond：
            需要等待的那个条件变量指针
        @mutex：
            线程互斥锁：为了保护cond所表示的那个事件/共享资源的。
            条件变量实际也是一个共享资源。
        @abstime：
            绝对时间，需要被唤醒的绝对时间
    */

    int pthread_cond_wait(pthread_cond_t *restrict cond,pthread_mutex_t
    *restrict mutex);
	/*
        @描述：
        	等待那个条件变量
        @cond：
        	需要等待的那个条件变量指针
        @mutex：
            线程互斥锁：为了保护cond所表示的那个事件/共享资源的。
            条件变量实际也是一个共享资源。
            
            mutex：locked 上锁
            pthread_cond_wait()
            {
                ... 准备工作
                mutex：unlock 解锁
                让出CPU 等待
                ...
                when 当条件产生的时候，其他的线程唤醒我的时候
                	mutex：locked 上锁
            }
        @return：
            成功返回0，被其他线程唤醒
            失败返回其他值
    */
```

### 唤醒线程/触发条件变量 `pthread_cond_broadcast`, `pthread_cond_signal`

````c++
NAME
    pthread_cond_broadcast, pthread_cond_signal 
    — broadcast or signal a condition (广播或发出条件信号)
    
SYNOPSIS

    #include <pthread.h>
    // 广播唤醒
    int pthread_cond_broadcast(pthread_cond_t *cond);
    /*
        @描述：
        	唤醒所有正在等待的线程
        @cond：
        	那个条件变量
        @return：
        	成功返回0，失败返回其他值
    */
	
    // 单个唤醒
    int pthread_cond_signal(pthread_cond_t *cond);
	/*
        @描述：
        	只唤醒一个线程在等待的线程。
        @cond：
        	那个条件变量
        @return：
        	成功返回0，失败返回其他值
    */
````

注意：广播唤醒和单个唤醒的区别 

- 广播唤醒：唤醒所有等待的线程，去执行任务，但是任务可能不够分，那么没分到的线程继续休眠 
- 单个唤醒：随机唤醒一个线程执行任务，其他线程继续休眠。

```c++
#include<iostream>
#include<unistd.h>
#include<pthread.h>
#include<queue>
#include<vector>

pthread_mutex_t mutex;
pthread_cond_t cond;
std::queue<std::string> groupWork;
bool should_exit = false;
int active_consumers = 10;  // 活跃消费者计数

void *createThread(void *args)
{
    std::cout<<"老板被创建"<<std::endl;
    usleep(1000);
    while(!should_exit)
    {
        std::string work;
        std::cout<<"请输入工作:";
        std::cin>>work;

        pthread_mutex_lock(&mutex);
        groupWork.push(work);//将工作添加到队列中
        std::cout<<"请输入群发/私发:";
        std::cin>>work;
        if(work == "私发")
        {
            pthread_cond_signal(&cond);
        }
        else if(work == "群发")
        {
            pthread_cond_broadcast(&cond);
        }
        else if(work == "退出")
        {
            should_exit = true;
            pthread_cond_broadcast(&cond);
            pthread_mutex_unlock(&mutex);
            break;
        }
        pthread_mutex_unlock(&mutex);
        usleep(1000);
    }
    return nullptr;
}

void *consumerThread(void* args)
{
    long thread_id = (long)args;
    std::cout<<"线程"<<thread_id<<"：被创建！"<<std::endl;
    
    while(1)
    {
        pthread_mutex_lock(&mutex);
        
        // 检查退出条件
        if(should_exit) {
            active_consumers--;
            pthread_mutex_unlock(&mutex);
            std::cout << "线程" << thread_id << "：收到退出信号，正常退出" << std::endl;
            break;
        }
        
        // 等待工作
        while(groupWork.empty() && !should_exit) {
            pthread_cond_wait(&cond, &mutex);
            std::cout << "线程" << thread_id << "被唤醒" << std::endl;
        }
        
        // 再次检查退出条件
        if(should_exit) {
            active_consumers--;
            pthread_mutex_unlock(&mutex);
            std::cout << "线程" << thread_id << "：收到退出信号，正常退出" << std::endl;
            break;
        }
        
        if(!groupWork.empty())
        {
            std::string work = groupWork.front();//将输入的工作即队首给到work
            groupWork.pop();//移除队首，即该工作，表示我做完了该工作
            pthread_mutex_unlock(&mutex);
            
            if(work == "退出")
            {
                pthread_mutex_lock(&mutex);
                should_exit = true;
                pthread_cond_broadcast(&cond);//当第一个线程被唤醒后，发现是退出信号，通知全部进程
                pthread_mutex_unlock(&mutex);
                std::cout << "线程" << thread_id << "：收到退出工作，通知其他线程退出" << std::endl;
                break;
            }
            
            std::cout<<"线程"<<thread_id<<"抢到["<<work<<"]工作，开始工作"<<std::endl;
            for(int i = 0;i < 5000;i++) usleep(2000);
            std::cout << "线程" << thread_id << ":工作完成，等待下一份工作!" << std::endl;
        }
        else
        {
            pthread_mutex_unlock(&mutex);
        }
    }
    return nullptr;
}

int main()
{
    pthread_t createTid;
    //std::vector<pthread_t> consumerTids(10);
    pthread_t consumerTids[10];

    pthread_mutex_init(&mutex,nullptr);
    pthread_cond_init(&cond,nullptr);
    
    pthread_create(&createTid,nullptr,createThread,nullptr);
    for(int i = 0;i<10;i++)
    {
        pthread_create(&consumerTids[i],nullptr,consumerThread,(void*)(long)i);
    }

    pthread_join(createTid,nullptr);
    for(int i = 0;i<10;i++)
    {
        pthread_join(consumerTids[i],nullptr);
    }

    pthread_mutex_destroy(&mutex);
    pthread_cond_destroy(&cond);
    
    std::cout << "所有线程已退出，程序正常结束" << std::endl;
    return 0;
}
```
