## 将两个源文件和一个头文件编译链接成一个可执行程序

假设有以下三个文件：

- **头文件`coordin.h`** 

```c++
#ifndef COORDIN_H_
#define COORDIN_H_

struct polar
{
    double distance; // distance from origin 与原点的距离
    double angle; // direction from origin 与原点的方向
};
struct rect
{
    double x; // horizontal distance from origin 从原点算起的水平距离
    double y; // vertical distance from origin 从原点算起的垂直距离
};

// prototypes
polar rect_to_polar(rect xypos);
void show_polar(polar dapos);

#endif
```

- 源文件:`file1.cpp`

```c++
// file1.cpp -- example of a three-file program 一个三文件程序的示例
#include <iostream>
#include "coordin.h" // structure templates, function prototypes
using namespace std;
int main()
{
    rect rplace;
    polar pplace;

    cout << "Enter the x and y values: ";//请输入x和y的值
    while (cin >> rplace.x >> rplace.y) // slick use of cin cin的巧妙运用
    {
        pplace = rect_to_polar(rplace);
        show_polar(pplace);
        cout << "Next two numbers (q to quit): "; //接下来的两个数字(输入q退出)
    }
    cout << "Bye!\n";
    return 0;
}
```

- 源文件:`file2.cpp`

```c++
// file2.cpp -- contains functions called in file1.cpp 包含在文件1中被调用
#include <iostream>
#include <cmath>
#include "coordin.h" // structure templates, function prototypes

// convert rectangular to polar coordinates 将直角坐标转换为极坐标
polar rect_to_polar(rect xypos)
{
    using namespace std;
    polar answer;

    answer.distance =
        sqrt( xypos.x * xypos.x + xypos.y * xypos.y);
    answer.angle = atan2(xypos.y, xypos.x);
    return answer; // returns a polar structure 返回一个极坐标结构体
}

// show polar coordinates, converting angle to degrees 
//显示极坐标，将角度转换为度数
void show_polar (polar dapos)
{
    using namespace std;
    const double Rad_to_deg = 57.29577951;

    cout << "distance = " << dapos.distance;
    cout << ", angle = " << dapos.angle * Rad_to_deg;
    cout << " degrees\n";
}
```
