#include <stdio.h>
#include <stdlib.h>
#define MAXSIZE 10
typedef struct LNode* List;
struct LNode
{
    int data[MAXSIZE];
    int Last;//最后一个下标位置，从0开始
};

struct LNode L;
List PtrL;
//访问下标为i的元素：L.data[i]或PtrL->data[i]
//线性表长度L.Last+1或PtrL->Last+1
//初始化
List MakeEmpty()
{
    List PtrL;
    PtrL = (List) malloc(sizeof(struct LNode));
    PtrL->Last = -1;
    return PtrL;
}
//查找
int Fin(int x,List PtrL)
{
    int i = 0;
    while(i<=PtrL->Last && PtrL->data[i]!= x)
    {
        i++;
    }
    if(i>PtrL->Last)    return -1;//如果循环到最后，还是没有找到返回-1
    else return i;//如果找到x 返回对应的下标
}
//插入
void Insert(int x,int i,List PtrL)//插入x在i号(i从1开始到n+1结束)位置，插在 PtrL链表上
{
    //插入位置:   1    2    ...    i     i+1   i+2 ...   n  n+1
    //下标：      0    1    ...   i-1     i    i+1 ...   n-1
    //已有数据:   a1    a2    ...  ai    ai+1  i+2 ...   an  ... -
    int j;
    if(PtrL->Last == MAXSIZE -1);//表空间已满，不能插入
    {
        printf("表满");
        return ;
    }
    if(i<1||i>PtrL->Last+2)
    {
        printf("位置不合法");
        return ;
    }
    //i：    1 2 3 4 5 
    //下标： 0 1 2 3
    //最多插入位置(从1开始)：list+1(表示在当前最后一个元素位置插入) list+2(表示在当前最后一个元素之后插入)
    for(j = PtrL->Last;j>= i-1;j--)
    {
        //i-1号 下标之后的数据，从最后一个开始，向后移动
        PtrL->data[j+1] = PtrL->data[j];
    }
    PtrL->data[i-1]= x;//将新元素值赋给i-1号下标位置
    PtrL->Last++;//Last仍然是最后一个元素的下标、
    return ;
}
    //删除
    //删除位置:   1    2    ...    i     i+1    i+2        ...   n  
    //下标：      0    1    ...   i-1     i     i+1     ...      n-1
    //已有数据:   a1    a2    ...  ai    ai+1   i+2       ...    an  ... -
void Insert(int x,int i,List PtrL)
{
    int j;
    if(i<1||i>PtrL->Last+1)
    {
        printf("不存在第%d个元素",i);
        return ;
    }
    for(j = i;j<=PtrL->Last;j++)
    {
        PtrL->data[i-1]=PtrL->data[i];
    }
    PtrL->data[j-1] = -1;
    PtrL->Last--;
    return ;
}
