# 1.两数之和

方法1：暴力破解，结果直接给到{i,j}

```c++
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        for (int i = 0; i < nums.size(); ++i) {
            for (int j = i + 1; j < nums.size(); ++j) {
                if (nums[i] + nums[j] == target) {
                    return {i,j};
                }
            }
        }
        return {};
    }
};
```

方法二：`哈希表，使用map将nums[i]作为key,i作为下标。`
`即a[nums[i]]=i  (这句话的意思是value = i,key = nums[i])`
`如果找到当前nums[i]所对应的target-nums[i]则将a[target-nums[i]]（key为target-nums[i]，所映射的value（下标））和 当前nums[i]的下标i作为vector<int>的两个结果返回`

````c++
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        unordered_map<int,int> Mymap;
        for(int i = 0;i<nums.size();i++)
        {
            if(Mymap.count(target-nums[i]))
            {
                return vector<int>{Mymap[target-nums[i]],i};
            }
            else
            {
                Mymap[nums[i]] = i;
            }
        }
        return vector<int>{};
    }
};
````



# 49.字母异位词分组

````c++
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <unordered_map>

using namespace std;

vector<vector<string>> groupAnagrams(vector<string>& strs) {
    unordered_map<string, vector<string>> map;
    
    // 构建哈希表，键是排序后的字符串
    for (string& s : strs) {
        string key = s;
        sort(key.begin(), key.end());  // 排序作为字母异位词的标识
        map[key].push_back(s);
    }
    
    // 将哈希表中的值转换为二维 vector
    vector<vector<string>> result;
    for (auto& pair : map) {
        result.push_back(pair.second);
    }
    
    return result;
}

int main() {
````

# 128.最长连续序列

- O（`n longn`）

```c++
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        vector<int> sorts = nums;
        sort(sorts.begin(),sorts.end());
        if(nums.empty())
            return 0;//判断是否为空
        int count = 1;
        int maxlen = 1;
        for(int i = 0;i<nums.size()-1;i++)
        {
            if (sorts[i] == sorts[i+1]) 
            {
                continue;
            }
            if(sorts[i]+1 == sorts[i+1])
            {
                count += 1;
                maxlen = max(maxlen,count);//更新最大值
            }
            else 
            {
                count = 1;//重置个数
            }
        }
        return maxlen;
    }
};
```

- O(n)

```c++
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        if(nums.empty())
            return 0;
        unordered_set<int> Myset(nums.begin(),nums.end());
        int maxlen = 0;
        for(int num:Myset)
        {
            // 只从序列的起点开始找
            if(Myset.find(num-1)==Myset.end())//未发现比当前数小1的数在set中，即这是该连续序列的首数字
            {
                int currentnum = num;
                int currentlen = 1
                //// 向后找连续的数字
                while(Myset.find(currentnum+1)!=Myset.end())
                {
                    currentnum++;//找到，更新当前数字，继续向后找
                    currentlen++;//找到，更新当前连续序列的长度
                }
                maxlen = max(maxlen,currentlen);//更新最大长度
            }
        }
        return maxlen;
    }
};
```

```c++
#include <iostream>
#include <list>
struct ListNode
{
    int val;
    ListNode* next;
    ListNode() : val(0), next(nullptr) {}
    ListNode(int x) : val(x), next(nullptr) {}
    ListNode(int x, ListNode *next) : val(x), next(next) {}
    ListNode(int x):val(x),next(NULL){}
};

class Solution
{
public:
    ListNode* addTwoNumbers(ListNode* l1,ListNode* l2)
    {
        ListNode* dummy = new ListNode(0);
        ListNode* current = dummy;
        int carry = 0;
        while(l1!=nullptr||l2!=nullptr||carry!=0)
        {
            int sum = carry;
            if(l1!=nullptr)
            {
                sum += l1->val;
                l1 = l1->next;
            }
            if(l2!=nullptr)
            {
                sum += l2->val;
                l2 = l2->next;
            }
            current->next = new ListNode(sum%10);
            current = current->next;
            carry = sum/10;
        }
        return dummy->next;
    }
};
```

## 滑动窗口

维护一个不包含重复字符的窗口（左指针`left`和右指针`right`），一个哈希集合(`unordered_set`）来记录当前窗口中的字符。一个int类型的`ans`用来记录最长无重复字符子串的长度

右指针`right`不断向右移动，如果当前字符`s[right]`不在集合中，就将其加入集合，并更新最长子串的长度。
如果当前字符`s[right]`已经在集合中，说明出现了重复字符，此时我们需要移动左指针`left`，并同时将左指针指向的字符从集合中移除，直到当前窗口中没有重复字符为止。
这样，当右指针遍历完整个字符串时，我们就得到了最长无重复字符子串的长度。

```c++
#include <iostream>
#include <string>
#include <unordered_set>
using namespace std;

class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_set <char> myset;
        int ans = 0;//用来记录最长无重复字符子串的长度
        int left = 0;
        for(int right = 0;right<s.size();right++)//右指针移动
        {
            while(myset.count(s[right]))//判断当前字符是否 存在于哈希表集合set
            {
                myset.erase(s[left]); //存在，则移除最左边的字符s[left]，缩小滑动窗口
                left += 1;//左指针偏移量+1
            }
            myset.insert(s[right]); //不存在重复字符，则存放到myset中
            ans = max(ans,right-left+1);//计算操作后的不含重复字符子串长度
        }
        return ans;
    }
};

int main()
{
    Solution sol;
    string s;
    cin >> s;
    cout << sol.lengthOfLongestSubstring(s) << endl;
    return 0;
}
```

## 算法特点

- **时间复杂度**：O(n)，每个字符最多被访问两次（右指针一次，左指针一次）
- **空间复杂度**：O(min(m,n))，其中m是字符集大小，n是字符串长度
- **优点**：一次遍历解决问题，效率高
- **适用场景**：需要找到满足某种条件的连续子串的问题

使用前缀和/哈希表解法

```c++
// prefix[j] - prefix[i-1] == k 从当前位置p(j) - p(i-1)看看有没有=k
// prefix[j] - k == prefix[i-1] 从当前位置p(j) - k 看看有没有前缀和=p(i-1)
```

哈希表中存放的是 	**键**:前缀和 **值**:前缀和存在的个数

当整数数组`nums`存放的是{1,2,3,4,1,11,-1,7,8,-46,10,10,10}，K为10时

**令：前缀和mp[0] = 1**

| :--- | :--- | :----------------------------------------------- | :------------- | :----------- | :-------------------- | :----------------- | :--------------------- |
| -1   | 0    | **p[j]-p[i-1]==k 当i为0时表示从0到j的前缀和为K** | **0**          | **×**        | **mp[0] = 1**         |                    | **mp[0] = 1**          |

| 1    | 2    | cur = 1+2 = 3                                    | 3              | 3-10 = -7    | 不存在                | count=0            | mp[3]=1                |

````c++
#include <iostream>
#include <memory>
#include <vector>
#include <unordered_map>
using namespace std;

// 给你一个整数数组 nums 和一个整数 k ，请你统计并返回 该数组中和为 k 的子数组的个数 。
// 子数组是数组中元素的连续非空序列。

//输入:nums = [1,2,3,4,1,11,-1,7,8,-46,10,10],k = 10;
//1234
//2341
//11 -1
//7 8 -46
//10
//1,2,3,4,1,11,-1,7,8,-46,10,10
//10 

//1 2 3 4 1 11 -1
//0 1 2 3 4 5   6
//[5,6]=[0,6]-[0-5-1]
//[0,4]=[0,4]-[0-0-1]
//令[-1] = 0
//前缀和：prefix[i] 表示从数组开头到索引 i 的所有元素之和。
//核心等式：子数组 [i..j] 的和 =（当 i>0）。
// 如果这个和等于 k，那么 prefix[i-1] = prefix[j] - k。

class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int,int> prefixCount;//键key:前缀和 值value:该前缀和存在的个数
        prefixCount[0]=1;
        int sum = 0,count = 0;
        for(int num:nums)
        {
            // i=j ;i<=j; i--;
            // prefix[j] - prefix[i-1] == k 从当前位置p(j) - p(i-1)看看有没有=k
            // prefix[j] - k == prefix[i-1] 从当前位置p(j) - k 看看有没有前缀和=p(i-1)
            sum += num;
            if(prefixCount.find(sum-k) != prefixCount.end()){
                count += prefixCount[sum-k];
                //count 加等于 前缀和为sum-k的个数;
            }
            //前缀和为sum的个数++; (键的值++)
            prefixCount[sum]++;
        }
        return count;
    }
};

int main()
{
    Solution s;
    vector<int> mynusms = {1,2,3,4,1,11,-1,7,8,-46,10,10,10};
    std::cout<<s.subarraySum(mynusms,10)<<std::endl;

    return 0;
}
````

## 标准库的`std::reverse`

reverse() 是 C++ 标准库提供的一个函数，定义在`<algorithm>`头文件中，用于**反转**一个序列的元素。

```c++
template< class BidirIt >
void reverse( BidirIt first, BidirIt last );//这是标准库的std::reverse
```

将范围 [`first`, `last`) 中的元素顺序反转。

 其行为等同于对每个整数 i 在 [`0`, `std::distance(first, last) / 2`) 范围内，将迭代器对 `first + i` 和 `(last - i) - 1` 应用 `std::iter_swap`

## 标准库的`std::swap`

标准库中的 swap 通常如下实现（C++11 及以后支持移动语义，这里给出简化版本）：

```c++
// 针对可移动类型的通用实现
template<class T>
void swap(T& a, T& b) {
    T temp = std::move(a);   // 移动构造 temp
    a = std::move(b);        // 移动赋值 a = b
    b = std::move(temp);     // 移动赋值 b = temp
}
```

对于像 int 这样的平凡类型，编译器通常会优化为更高效的直接内存交换，甚至生成 xchg 指令。

- **方法**:
  - 第一步:翻转整个数组
  - 第二步:从第k个元素后,将数组划分为左右两块子数组
  - 第三步:左右两块子数组,各自翻转
  - 第四步:完成

```c++
#include <iostream>
#include <vector>
using namespace std;

class Solution {
public:
    void reverse(vector<int>& nums,int start,int end){
        while(start<end){
            swap(nums[start],nums[end]);
            start++;
            end--;
        }
    }
    void rotate(vector<int>& nums, int k) {
        k %= nums.size();
        reverse(nums,0,nums.size()-1);
        reverse(nums,0,k-1);
        reverse(nums,k,nums.size()-1);
    }
};  

int main(){
    vector<int> nums = {1,2,3,4,5,6,7};
    Solution s;
    int k = 10;
    s.rotate(nums,k);
    for(int i : nums){
        std::cout<<i<<" ";
    }
    return 0;
}
```

```c++
        int length = nums.size();
        vector<int> answer(length,0);
        vector<int> left(length,0);//left[i] 为 i左侧所有元素的乘积和
        vector<int> right(length,0);//right[i] 为 i右侧所有元素的乘积和
        left[0] = 1;//如果最左侧元素,i的左侧没有元素,它的乘积和为1
        for(int i = 1;i<length;i++){//从第1个元素开始,计算i左侧的前缀积
            left[i] = left[i-1] * nums[i-1];
        }
        right[length-1] = 1;//如果是最右侧元素,i的右侧没有元素,它的乘积为1
        for(int i = length-2;i>=0;i--){//从最右侧-1的元素开始,计算i右侧的后缀积
            right[i] = right[i+1] * nums[i+1];
        }
        for(int i = 0;i<length;i++){
            answer[i] = left[i] * right[i];//答案为i左侧积和右侧积相乘,即不算当前的第i位置数组的乘积
        }
        return answer;
    }
};

int main(){

    vector<int> nums = {4,5,1,8,2};
    Solution s;
    vector<int> answer = s.productExceptSelf(nums);
    for(int i : answer){
        std::cout<<i<<" ";
    }
    
    
    return 0;
}
```

```c++
#include <iostream>
#include <vector>
using namespace std;

class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        vector<int>ans;
//初始化前缀和为1
//nums:  4    5         1           8           2 
//ans:   1    1*4       1*4*5       1*4*5*1     1*4*5*1*8
//result 1*4  1*4*5     1*4*5*1     1*4*5*1*8   1*4*5*1*8*2(×不需要了)
        int result = 1;
        for(int i = 0; i < nums.size(); i++){//计算前缀积
            ans.push_back(result);//初始化前缀积
            result = result * nums[i];//每一次循环,result记录为下一次的前缀积
        }

//恢复result=1,即后缀和初始化为1;在这里ans[i]表示上一次该i位置的前缀积
//nums:   4                        5                 1                8               2 
//ans:    ans[i]*5*1*8*2*1         ans[i]*1*8*2*1    ans[i]*8*2*1     ans[i]*2*1      ans[i]*1

        result = 1;

            result *= nums[i];//计算后缀积
        }
        return ans;
    }
```

# 矩阵置零

## 法1,暴力

> 找出原二维数组中元素值为0的下标位置记为`zero_i`,`zero_j`并插入到std::set<std::pair<int, int>> zeros中,然后遍历原二维数组,将下标为`zero_i`的行全置为零,为`zero_j`的列全置为零

```c++
#include <iostream>
#include <vector>
#include <set>
#include <utility>
using namespace std;

class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        if (matrix.empty() || matrix[0].empty()) 
            return;
        int m = matrix.size();
        int n = matrix[0].size();

        std::set<std::pair<int, int>> zeros;
        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                if (matrix[i][j] == 0) {//原二维数组中值为0的下标位置插入到zeros
                    zeros.insert({i, j});
                }
            }
        }

        for(const auto& pos : zeros){
            int row = pos.first;
            int col = pos.second;
            for(int j = 0;j<n;j++){
                matrix[row][j]=0;//该元素的列置零
            }
            for(int i = 0;i<m;i++){
                matrix[i][col]=0;//该元素的行置零
            }
        }
    }
};

int main(){
    Solution s;
    vector<vector<int>> matrix={{0,1,2,0},{3,4,5,2},{1,3,1,5}};
    s.setZeroes(matrix);
    std::cout<<"[";
    for(int i = 0;i<matrix.size();i++){
        std::cout<<"[";
        for(int j = 0;j<matrix[0].size();j++){
            std::cout<<matrix[i][j];
            if(j != matrix[0].size()-1){
                std::cout<<",";
            }
        }
        std::cout<<"]";
        if(i != matrix.size()-1){
            std::cout<<",";
        }
    }
    std::cout<<"]";

    return 0;
}
```

## 法2,用第0行和第0列作为标记

> 1. 首先，**进行初始化**，m代表行数,n代表列数;用两个布尔值表示第0行和第0列是否有0元素，默认false表示没有0元素
>
> 2. 然后，使用**循环判断第0行和第0列是否有0元素**,如果第0行有0元素，则将`firstRowzero`置为true。同理，如果第0列有0元素，则将`firstColzero`置为true。其中：
>    - 第0行的 所有元素 的表示方式是：`matrix[0][j]`（j为列数，从0到n）
>    - 第0列的 所有元素 的表示方式是：`matrix[i][0]`（i为行数，从0到m）
> 3. 再从下标（1,1）位置开始遍历，如果找到某位置（i，j）元素值为0，则将它的行首和列首归0
> 4. 根据第三步,从下标（1,1）位置开始遍历,如果该位置（i，j）的行首或列首值为0,则将它也归0
> 5. 最后，如果第二步判断出
>    - 第0行有0元素，则将第0行的所有元素归0
>    - 第0列有0元素，则将第0列的所有元素归0

```c++
#include <iostream>
#include <vector>
#include <set>
#include <utility>
using namespace std;

class Solution {
public:
// [ [1, 2, 3],
//   [4, 0, 6],
//   [7, 8, 9] ]
    void setZeroes(vector<vector<int>>& matrix) {
        int m = matrix.size();     //m行,有m行
        int n = matrix[0].size();  //n列,有n列
        bool firstRowzero = false;
        bool firstColzero = false;

        //判断第0行和第0列有没有0元素
        for(int j = 0;j<n;j++){    //第0行,一行有n列个元素
            if(matrix[0][j]==0){
                firstRowzero = true;//如果第0行有元素值为0,则标记需要处理
            }
        }
        for(int i = 0;i<m;i++){     //第0列,一列有m行个元素
            if(matrix[i][0]==0){ //如果第0列有元素值为0,则标记需要处理
                firstColzero = true;
            }
        }
        
        //用第0行和第0列作为标记,从第1行和第1列开始,如果有元素为0
        //则将其对应的 行首 和 列首 归0
        for(int i = 1;i<m;i++){
            for(int j = 1;j<n;j++){
                if(matrix[i][j]==0){ //如果该元素为0
                    matrix[i][0]=0;  //将该行首的元素归0
                    matrix[0][j]=0;  //将该列首的元素归0
                }
            }
        }

        //根据标记进行矩阵置零
        for(int i = 1;i<m;i++){
            for(int j = 1;j<n;j++){
                if(matrix[i][0]==0 || matrix[0][j]==0){//如果该元素的行首或列首值为0
                    matrix[i][j] = 0;//将该元素归0
                }
            }
        }

        //处理第0行和第0列
        if(firstRowzero){//处理第0行,如果
            for(int j = 0;j<n;j++){
                matrix[0][j] = 0;
            }
        }
        if(firstColzero){//处理第0列
            for(int i = 0;i<m;i++){
                matrix[i][0] = 0;
            }
        }
    }
};

int main(){
    Solution s;
    vector<vector<int>> matrix={{0,1,2,0},{3,4,5,2},{1,3,1,5}};
    s.setZeroes(matrix);
    std::cout<<"[";
    for(int i = 0;i<matrix.size();i++){
        std::cout<<"[";
        for(int j = 0;j<matrix[0].size();j++){
            std::cout<<matrix[i][j];
            if(j != matrix[0].size()-1){
                std::cout<<",";
            }
        }
        std::cout<<"]";
        if(i != matrix.size()-1){
            std::cout<<",";
        }
    }
    std::cout<<"]";

    return 0;
}
```

```c++
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        vector<int> myvector = {};
        //设置上下左右边界
        if(matrix.empty())
            return myvector;
        int up = 0;//上边界
        int down = matrix.size()-1;//下边界
        int left = 0;//左边界
        int right = matrix[0].size()-1;//右边界

            for(int i = left;i<=right;i++)

                myvector.push_back(matrix[i][right]);//向下移动

                myvector.push_back(matrix[i][left]);//向上移动
            if(++left > right) break;//更新左边界
        }
        return myvector;
    }
```

## 法1:使用辅助数组

对于矩阵中第 i 行的第 j 个元素，在旋转后，它出现在倒数第 i 列的第 j 个位置。

- `matrix[i][j]，在旋转后，它的新位置为 matrix[j][n-i-1]；`

使用一个与matrix大小相同的辅助数组ans,存储旋转后的结果,遍历每一个matrix元素,运用以下规则存放到 ans中。遍历完成后将ans中的结果复制到原数组matrix中。

- `ans[j][n-i-1]=matrix[i][j];`	将右边的值赋值到左边；

```c++
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        vector<vector<int>> ans = matrix;
        int n = matrix.size();
        for(int i = 0;i<n;i++){
            for(int j = 0;j<n;j++){
                ans[j][n-i-1]=matrix[i][j];
            }
        }
        matrix = ans;
        // i行变n-1-i列
        // j列变行
    }
};
```

## 法2:就地旋转(就地交换元素后翻转每一行元素)

将`matrix[i][j]`元素与`matrix[j[i]`互换位置,然后将每一行的元素进行翻转

```c++
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n=matrix.size(), m=matrix[0].size();

        for(int i=0;i<n;i++)
            for(int j=i;j<m;j++)
                swap(matrix[i][j],matrix[j][i]);

        for(int i=0;i<n;i++)
            reverse(matrix[i].begin(),matrix[i].end());
    }
};
```

- 二分查找

```c++
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        for(vector<int>& row : matrix){//循环变量matrix二维数组的每一行
            //对每一行都使用一次二分查找,使用lower_bound查找第一个大于等于target的元素
            std::vector<int>::iterator lower = std::lower_bound(row.begin(),row.end(),target);
            if(lower!=row.end() && *lower==target)//如果找到并且该位置正好是target值则返回 真
                return true;
        }
        return false;
    }
};
```

# 相交链表

哈希表、链表、双指针

## 方法一:哈希表

```c++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        unordered_set<ListNode*> myset;

        ListNode* cur = headA;
        while(cur){
            myset.insert(cur);
            cur = cur->next;
        }

        cur = headB;
        while(cur){
            if(myset.find(cur) != myset.end()){
                //cur 是一个 ListNode* 类型的指针，指向相交节点
                return cur;//返回当前链表位置,之后程序通过cur->val得到当前链表对应的值
            }
            cur = cur->next;
        }
        return nullptr;
    }
};
```

**复杂度分析**：

- 时间复杂度：O(m+n)，其中 m 和 n 是分别是链表 headA 和 headB 的长度。需要遍历两个链表各一次。

- 空间复杂度：O(m)，其中 m 是链表 headA 的长度。需要使用哈希集合存储链表 headA 中的全部节点。

## ==方法二：双指针==

使用双指针的方法，可以将空间复杂度降至 *O*(1)。
