## 11.2.1 基于升序链表的定时器

> 代码清单 11-2 升序定时容器链表

```c
#ifndef LIS_TIMER
#define LIS_TIMER

#include <time.h>
#include <netinet/in.h>
#include <stddef.h>
#include <stdio.h>
#define BUFFER_SIZE 64
class util_timer;//前向声明

//用户数据结构，客户端socket地址、socket文件描述符、读缓存和定时器
struct client_data
{
    sockaddr_in address;
    int sockfd;
    char buf[BUFFER_SIZE];
    util_timer* timer;
};

//定时器类
class util_timer
{
public:
    util_timer():prev(NULL),next(NULL){}
public:
    time_t expire;//任务的超时时间，这里使用绝对时间
    void(*cb_func)(client_data*);//任务回调函数
    //回调函数处理的客户数据，由定时器的执行者传递给回调函数
    client_data* user_data;
    util_timer* prev;//指向上一个定时器
    util_timer* next;//指向下一个定时器
};

class sort_timer_list
{
public:
    sort_timer_list():head(NULL),tail(NULL){}
    //链表被销毁，删除其中所有的定时器。
    ~sort_timer_list()
    {
        util_timer* tmp = head;
        while(tmp)
        {
            head = tmp->next;
            delete tmp;
            tmp = head;
        }
    }
    //将目标定时器timer添加到链表中。
    void add_timer(util_timer* timer)
    {
        if(!timer)
        {
            return;
        }
        if(!head)
        {
            head = tail = timer;
            return;
        }
        //如果目标定时器的超时时间小于当前链表中所有定时器的超时时间，则把该定时器插入链表头部，作为链表新的头结点。
        //否则就需要调用重载函数add_timer();
        //把它插入链表中合适的位置，以保证链表的升序特性
        if(timer->expire < head->expire)
        {
            timer->next = head;
            head->prev = timer;
            head = timer;
            return;
        }
        add_timer(timer,head);//该函数表示将目标定时器timer添加到head之后的部分链表中
    }
    //当某个定时任务发生变化时，调整对应的定时器在链表中的位置。这个函数只考虑被调整的定时器的超时时间延长的情况
    //即该定时器需要往链表的尾部移动。
    void adjust_timer(util_timer* timer)
    {
        if(!timer)
        {
            return;
        }
        util_timer* tmp = timer->next;
        //如果被调整的目标定时器处在链表尾部，或者  该定时器新的超时值仍然小于其下一个定时器的超时值，则不用调整
        if(!tmp || (timer->expire < tmp->expire))
        {
            return;
        }
        //如果目标定时器是链表的头部，则将该定时器从链表中取出并重新插入链表
        if(timer == head)
        {
            head = head->next;
            head->prev = NULL;
            timer->next = NULL;
            add_timer(timer,head);//该函数表示将目标定时器timer添加到head之后的部分链表中
        }
        //如果目标定时器不是链表的头节点，则将该定时器从链表中取出，然后插入其原来的位置之后的部分链表中。
        else
        {
            timer->prev->next = timer->next;
            timer->next->prev = timer->prev;
            add_timer(timer,timer->next);//该函数表示将目标定时器timer添加到 timer->next之后的部分链表中
        }
    }
    //将目标定时器timer从链表中删除
    void del_timer(util_timer* timer)
    {
        if(!timer)
        {
            return;
        }
        //下面这个条件成立表示链表中只有一个定时器，即目标定时器
        if((timer == head) && (timer == tail))
        {
            delete timer;
            head = NULL;
            tail = NULL;
            return;
        }
        //如果链表中至少有两个定时器，且目标定时器是链表的头结点，则将链表的头结点重置为原头结点的下一个节点，然后删除目标定时器
        if(timer == head)
        {
            head = head->next;
            head->prev = NULL;
            delete timer;
            return;
        }
        //如果链表中至少有两个定时器，且目标定时器是链表的尾结点，则将链表的尾节点重置为原尾节点的上一个节点，然后删除目标定时器
        if(timer == tail)
        {
            tail = tail->prev;
            tail->next = NULL;
            delete timer;
            return;
        }
        //如果目标定时器位于链表的中间，则把他前后的定时器串联起来，然后删除目标定时器
        timer->prev->next = timer->next;
        timer->next->prev = timer->prev;
        delete timer;
    }
    //SIGALRM信号每次被触发就在其信号处理函数(如果使用同一事件源，则是主函数)中执行一次tick函数，以处理链表上的到期任务
    void tick()
    {
        if(!head)
        {
            return;
        }
        printf("timer tick\n");
        time_t cur = time(NULL);//获取系统当前时间
        util_timer* tmp = head;
        //从头节点开始处理每个定时器，直到遇到一个尚未到期的定时器，这就是定时器的核心逻辑
        while(tmp)
        {
            //因为每个定时器都使用绝对时间作为超时值，所以我们可以把定时器的超时值和系统当前时间比较，来判断定时器是否到期。
            if(cur < tmp->expire);
            {
                break;
            }
            //调用定时器的回调函数，以执行定时任务
            tmp->cb_func(tmp->user_data);
            //执行完定时器中的定时任务之后，就将它从链表中删除，以重置链表头结点
            head = tmp->next;
            if(head)
            {
                head->prev = NULL;
            }
            delete tmp;
            tmp = head;
        }
    }
private:
    //一个重载的辅助函数，它被公有的add_timer函数和adjust_timer函数调用，该函数表示将目标定时器timer添加到lst_head之后的部分链表中
    void add_timer(util_timer* timer,util_timer* lst_head)
    {
        util_timer* prev = lst_head;
        util_timer* tmp = prev->next;
        //遍历lst_head节点之后的部分链表，直到找到一个超时时间大于目标定时器的超时时间的节点，并将目标定时器插入该节点之前
        while(tmp)
        {
            if(timer->expire < tmp->expire)
            {
                prev->next = timer;
                timer->next = tmp;
                timer->prev = prev;
                tmp->prev = timer;
                break;
            }
            prev = tmp;
            tmp = tmp->next;
        }
        //如果遍历完lst_head节点之后的部分链表，仍为找到超时时间大于目标节点的超时时间的节点，
        //则将目标定时器插入链表尾部，并把它设置为链表新的尾节点
        if(!tmp)
        {
            prev->next =timer;
            timer->prev = prev;
            timer->next = NULL;
            tail = timer;
        }
    }
private:
    util_timer* head;
    util_timer* tail;
};

#endif
```

这段代码实现了一个**升序双向链表定时器管理器**，用于管理多个定时器，并按超时时间（`expire`）升序排列。它的核心组件包括：

- `client_data`：存储客户端连接信息（地址、socket fd、缓冲区）以及指向其定时器的指针。
- `util_timer`：定时器类，包含超时时间（绝对时间）、回调函数指针、客户数据指针，以及前后指针。
- `sort_timer_list`：定时器链表类，提供添加、删除、调整定时器的方法，以及`tick()`函数处理到期的定时器。

1. **数据结构**

- `client_data` 将连接与定时器绑定，方便回调函数获取连接信息。
- `util_timer` 采用双向链表节点设计，便于插入和删除。回调函数`cb_func`接收`client_data*`，可在定时器到期时执行具体操作（如关闭连接）。

2. **链表操作**

- **添加定时器**：`add_timer` 保持链表升序。若新定时器超时时间小于头结点，则插入头部；否则调用私有辅助函数`add_timer(timer, head)`，从`head`开始遍历找到合适位置插入。
- **调整定时器**：`adjust_timer` 用于处理超时时间**延长**的情况（例如连接有活动，需推迟关闭）。它会检查定时器是否需要后移：如果当前定时器超时值仍小于下一个，则无需调整；否则将其从链表中取出，并重新插入到合适位置（从原下一个节点开始查找）。
- **删除定时器**：`del_timer` 根据位置（头、尾、中间）更新前后指针，并`delete`定时器对象。
- **心跳函数**：`tick()` 遍历链表，检查到期的定时器，执行回调并删除它们。

3. **`tick`函数**

- `tick` 函数是整个定时器管理器的心脏，它负责**定期扫描链表，找出所有已经到期的定时器，并执行它们预设的回调函数**。简单来说，它就是那个不断“滴答”的时钟，每到一定时刻就检查一下：“有哪些闹钟已经响过了？该处理它们了。”
- **线程安全**：若`tick()`在信号处理函数中执行，而主循环可能同时操作链表（添加/调整/删除），则需加锁保护，或将定时事件统一处理（如通过管道将信号通知主循环，在主循环中执行`tick()`）。
- **定时精度**：`tick()`调用的频率决定了检测精度，通常与超时时间相比可以粗一些（例如每1秒一次）。
- **回调函数设计**：回调函数应只做关闭连接、释放资源的操作，避免长时间阻塞。若需要更复杂的逻辑，可考虑通知主循环处理。
- **链表操作效率**：对于高并发连接（数万），O(n)的插入/调整可能成为瓶颈，此时可考虑使用时间轮或最小堆优化。但本实现简单直观，适合学习或中小规模应用。

### **内存管理**：

`del_timer`中直接`delete`定时器，如果回调函数中已经`delete`了`client_data`，则需确保`user_data`指针不会被二次释放。通常回调函数只关闭socket，释放`client_data`由其他机制负责，或定时器删除时一并释放。设计时需明确所有权。

### 它解决了什么问题？

在编写网络服务器（例如HTTP服务器、聊天服务器）时，通常会遇到这样的需求：
服务器同时和成千上万个客户端保持连接，但有些客户端可能因为网络故障或用户离开，长时间不发送任何数据。如果一直保留这些连接，会浪费服务器的内存、文件描述符等宝贵资源。所以需要一种机制**定期清理掉那些长时间无活动的连接**，这就是“处理非活动连接”。

如何实现呢？可以为每个新连接的客户端设置一个定时器（闹钟），超时时间设为比如30秒。如果在这30秒内，该客户端发送了数据（有活动），我们就**重新调整它的定时器**，把超时时间往后推迟30秒（相当于重新设置闹钟）。如果一直没活动，那么30秒后定时器就会“响铃”，执行回调函数——这个回调函数的内容就是**关闭这个客户端连接**，并释放相关资源。

而上面那段代码中的`sort_timer_list`，正好可以用来高效地管理这些成千上万个定时器。它用升序链表保证：

- 每次只需要检查链表头部的定时器就知道最早到期的任务是什么。
- 添加、删除、调整定时器只需要遍历链表找到合适位置（O(n)复杂度，对于中小规模连接足够了）。
- `tick`函数可以定期被触发（比如每1秒一次），快速找出所有到期连接并清理。

因此，这段代码**解决了在大量并发连接中高效管理超时任务的问题**，是非活动连接检测机制的经典实现基础。

### 在非活动连接处理中的应用

服务器需要定期清理长时间无数据交互的客户端连接，避免资源浪费。利用上述定时器链表的典型流程如下：

#### 1. 初始化

创建`sort_timer_list`对象，并在程序主循环中设置定时信号（如`SIGALRM`），每隔固定时间（例如5秒）触发一次`tick()`。

#### 2. 为新连接创建定时器

当接受新连接时：

- 分配`client_data`，填充`sockfd`、地址等。
- 创建`util_timer`对象：
  - `expire = time(NULL) + TIMEOUT`（超时时间，例如30秒）。
  - `cb_func` 设置为关闭连接的回调函数（如`close_conn`），该函数接收`client_data*`，负责关闭socket、释放资源等。
  - `user_data` 指向该连接的`client_data`。
- 将`client_data->timer`指向此定时器，以便后续调整。
- 调用`sort_timer_list.add_timer(timer)`加入链表。

#### 3. 连接活动时延长超时

当在连接上成功收发数据（例如`read`/`write`返回）时：

- 获取该连接对应的定时器`timer`。
- 更新`timer->expire = time(NULL) + TIMEOUT`（新的超时时间）。
- 调用`adjust_timer(timer)`调整其在链表中的位置（可能后移）。

#### 4. 定期检测到期连接

在定时信号处理函数或主循环中调用`tick()`：

- 获取当前时间`now`。
- 遍历链表，对每个定时器检查`if (now >= timer->expire)`，若是则：
  - 执行`timer->cb_func(timer->user_data)`（回调会关闭连接）。
  - 从链表中删除该定时器（注意回调中通常也会将`client_data->timer`置NULL，避免后续误用）。
- 遇到第一个未到期的定时器即可停止，因为链表是有序的。

#### 5. 关闭连接时清理定时器

如果主动关闭连接（如服务器决定关闭），应调用`del_timer`从链表中移除对应的定时器，并释放定时器对象。注意要先将`client_data->timer`设为NULL，避免悬空指针。

### 实际应用中的流程

1. **新连接到达**
   - 创建`client_data`对象，保存连接信息。
   - 创建一个`util_timer`，设置超时时间（`expire = now + 30秒`），回调函数设为“关闭连接”的函数。
   - 调用`add_timer`把定时器加入链表。
2. **连接有数据收发**
   - 从该连接的`client_data`中找到它的定时器。
   - 更新它的`expire = now + 30秒`。
   - 调用`adjust_timer`调整它在链表中的位置（可能往后移动）。
3. **定时检查（tick）**
   - 程序每隔一段时间（如1秒）调用一次`tick`。
   - `tick`从链表头部开始，把当前时间之前的所有定时器依次取出，执行回调函数（关闭对应连接），并删除定时器。
4. **连接主动关闭**
   - 如果服务器主动关闭某个连接（比如收到客户端正常关闭的信号），则调用`del_timer`从链表中删除对应的定时器，避免重复清理。

通过这种方式，服务器就能自动维护那些活跃的连接，同时及时释放不再活跃的连接所占用的资源。

### 总结

该升序定时器链表为处理非活动连接提供了一个基础框架：通过定时器关联连接，定期扫描到期连接并关闭。使用时只需将每个新连接加入链表，活动时调整超时，并在心跳中执行`tick()`即可。尽管实现简单，但需要处理好线程安全、时间获取等问题，并根据实际场景考虑扩展性。

### 执行效率

添加定时器的时间复杂度是O(N)，删除定时器的复杂度是O(1),执行定时任务的时间复杂度是O(1)。

## 11.2.2 处理非活动连接

> 代码清单11-3 关闭非活动连接

```c
#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <assert.h>
#include <stdio.h>
#include <signal.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <fcntl.h>
#include <stdlib.h>
#include <sys/epoll.h>
#include <pthread.h>
#include <libgen.h>
#include "11-2升序定时器链表.h"

#define FD_LIMIT 65535
#define MAX_EVENT_NUMBER 1024
#define TIMESLOT 5

static int pipefd[2];
//利用代码清单11-2中的升序定时器链表来管理定时器
static sort_timer_list timer_list;//定时器链表对象
static int epollfd = 0;

int setnonblocking(int fd)
{
    int old_option = fcntl(fd,F_GETFL);
    int new_option = old_option | O_NONBLOCK;
    fcntl(fd,F_SETFL,new_option);
    return old_option;
}

void addfd(int epollfd,int fd)
{
    epoll_event event;
    event.data.fd = fd;
    event.events = EPOLLIN | EPOLLET;
    epoll_ctl(epollfd,EPOLL_CTL_ADD,fd,&event);
    setnonblocking(fd);
}

void sig_handler(int sig)//信号的处理
{
    int save_errno = errno;
    int msg = sig;
    send(pipefd[1],(char*)&msg,1,0);//向管道中写信号
}

void addsig(int sig)
{
    struct sigaction sa;
    memset(&sa,'\0',sizeof(sa));
    sa.sa_handler = sig_handler;
    sa.sa_flags |= SA_RESETHAND;
    sigfillset(&sa.sa_mask);
    assert(sigaction(sig,&sa,NULL) != -1);
}

void timer_handler()
{
    //定时处理任务，实际上就是调用tick()函数
    timer_list.tick();
    //因为一次alarm调用只会引起一次SIGALRM信号，所以需要重新定时，以不断地触发SIGALRM信号
    alarm(TIMESLOT);
}

//定时器回调函数，它删除非活动连接的socket上的注册事件，并关闭
void cb_func(client_data* user_data)
{
    epoll_ctl(epollfd,EPOLL_CTL_DEL,user_data->sockfd,0);
    assert(user_data);
    close(user_data->sockfd);
    printf("close fd %d\n",user_data->sockfd);
}

int main(int argc,char* argv[])
{
    if(argc<=2)
    {
        printf("Usage: %s ip_address port_number\n", basename(argv[0]));
        return 1;
    }
    const char *ip = argv[1];
    int port = atoi(argv[2]);

    int ret = 0;
    struct sockaddr_in address;
    bzero(&address,sizeof(address));
    address.sin_family = AF_INET;
    inet_pton(AF_INET,ip,&address.sin_addr);
    address.sin_port = htons(port);

    int listenfd = socket(PF_INET,SOCK_STREAM,0);
    assert(listenfd >= 0);

    ret = bind(listenfd,(struct sockaddr*)&address,sizeof(address));
    assert(ret != -1);

    ret = listen(listenfd,5);
    assert(ret != -1);

    epoll_event events[MAX_EVENT_NUMBER];
    int epollfd = epoll_create(5);
    assert(epollfd != -1);
    addfd(epollfd,listenfd);

    ret = socketpair(PF_UNIX,SOCK_STREAM,0,pipefd);
    assert(ret != -1);
    setnonblocking(pipefd[1]);
    addfd(epollfd,pipefd[0]);

    //设置信号处理函数
    addsig(SIGALRM);
    addsig(SIGTERM);
    bool stop_server = false;

    client_data* users = new client_data[FD_LIMIT];
    bool timeout = false;
    alarm(TIMESLOT);//定时

    while(!stop_server)
    {
        int number = epoll_wait(epollfd,events,MAX_EVENT_NUMBER,-1);
        if((number<0)&&(errno != EINTR))
        {
            printf("epoll failure\n");
            break;
        }
        for(int i = 0;i<number;i++)
        {
            int sockfd = events[i].data.fd;
            //处理新到的客户连接
            if(sockfd == listenfd)
            {
                struct sockaddr_in client_address;
                socklen_t client_addresslength = sizeof(client_address);
                int connfd = accept(listenfd,(struct sockaddr*)&client_address,&client_addresslength);
                addfd(epollfd,connfd);
                users[connfd].address = client_address;
                users[connfd].sockfd = connfd;
                //创建定时器，设置其回调函数与超时时间，然后绑定定时器与用户数据，最后将定时器容器添加到链表timer_list中
                util_timer* timer = new util_timer;
                timer->user_data = &users[connfd];
                timer->cb_func = cb_func;
                time_t cur = time(NULL);
                timer->expire = cur + 3*TIMESLOT;
                users[connfd].timer = timer;
                timer_list.add_timer(timer);
            }
            //处理信号
            else if((sockfd == pipefd[0]) && events[i].events & EPOLLIN)
            {
                int sig;
                char signal[1024];
                ret = recv(pipefd[0],signal,sizeof(signal),0);
                if(ret == -1)
                {
                    //handle ther error
                    continue;
                }
                else if(ret == 0)
                {
                    continue;
                }
                else 
                {
                    for(int i = 0;i<ret;i++)
                    {
                        switch(signal[i])
                        {
                            case SIGALRM:
                            {
                                // 使用timeout变量标记有定时任务需要处理，但是不立即处理定时任务，
                                // 因为定时任务的优先级不是很高，优先处理其他更重要的任务
                                timeout = true;
                                break;
                            }
                            case SIGTERM:
                            {
                                stop_server = true;
                            }
                        }
                    }
                }
            }
            //处理客户连接上收到的数据
            else if(events[i].events & EPOLLIN)
            {
                memset(users[sockfd].buf,'\0',BUFFER_SIZE);
                ret = recv(sockfd,users[sockfd].buf,BUFFER_SIZE-1,0);
                printf("get %d bytes of client data %s from %d \n",ret,users[sockfd].buf,sockfd);

                util_timer* timer = users[sockfd].timer;
                if(ret<0)
                {
                    //如果发生读错误，则关闭连接，并移除其对应的定时器
                    if(errno != EAGAIN)//EAGAIN表示：资源暂时不可用，操作会阻塞
                    //非阻塞模式下，读无数据、写缓冲区满、`accept` 无新连接、`connect` 未完成等。
                    {
                        cb_func(&users[sockfd]);
                        if(timer)
                        {
                            timer_list.del_timer(timer);
                        }
                    }
                    else if (ret == 0)
                    {
                        //如果对方关闭了连接，则我们也关闭连接，并移除其对应的定时器
                        cb_func(&users[sockfd]);
                        if(timer)
                        {
                            timer_list.del_timer(timer);
                        }
                    }
                    else 
                    {
                        //如果某个客户端上有数据可读，则需要调整该连接对应的定时器，来延迟该连接被关闭的时间
                        if(timer)
                        {
                            time_t cur = time(NULL);
                            timer->expire = cur + 3* TIMESLOT;
                            printf("adjust timer once\n");
                            timer_list.adjust_timer(timer);
                        }
                    }
                }
            }
            else 
            {
                //others
            }
        }
        //最后处理定时事件，因为I/O事件具有更高的优先级，这样会导致定时任务不能精准地按照预期的时间执行
        if(timeout)
        {
            timer_handler();
            timeout = false;
        }
    }
    close(listenfd);
    close(pipefd[1]);//先关闭写端
    close(pipefd[0]);//再关闭读端
    delete []users;
    return 0;
}
```

# 11.3 I/O复用系统调用的超时参数

> 代码清单11-4 利用I/O复用系统调用定时

```c
#define TIMEOUT 5000 // 总的定时周期，例如5秒

int main() {
    int timeout = TIMEOUT; // 当前需要等待的超时时间
    time_t start, end;

    while (1) {
        start = time(NULL); // 1. 记录开始等待的时间点

        // 2. 调用 epoll_wait，使用当前计算出的 timeout 值
        int event_count = epoll_wait(epollfd, events, MAX_EVENT_NUM, timeout);
		if (number < 0 && errno != EINTR) //将EINTR被信号中断作为非致命错误，
        {//如果是其他errno则 认为是 致命错误，结束执行。
            printf("epoll failure\n");
            break;
        }
        
        end = time(NULL);   // 3. 记录返回时的时间点

        if (event_count == 0) {
            // 情况A：超时时间到，没有I/O事件
            // 执行定时任务函数...
            printf("定时任务触发\n");
            timeout = TIMEOUT; // 重置超时时间，准备下一个周期
            continue;
        }

        // 情况B：有I/O事件发生
        // 处理所有I/O事件...

        // 4. 核心调整：计算本次 epoll_wait 实际持续了多久
        int elapsed_ms = (end - start) * 1000; 
        // 从总的超时时间里减去已经过去的时间
        timeout -= elapsed_ms; //动态调整 timeout 参数。

        // 5. 检查剩余时间
        if (timeout <= 0) {
            // 这意味着在处理I/O事件的过程中，定时任务其实已经到期了
            // 需要立即执行定时任务，并重置超时时间
            // 执行定时任务函数...
            printf("在处理I/O时定时器已到期，立即执行\n");
            timeout = TIMEOUT;
        }
        // 如果 timeout > 0，下一次 epoll_wait 会等待剩余的毫秒数
    }
    return 0;
}
```

书中清单11-4之所以要动态计算并调整 `epoll_wait` 的超时参数，其核心目的是为了实现**精确的定时器管理**，确保定时任务能在预定的绝对时间点被唤醒并执行，同时还能动态适应每次循环处理耗时，提高CPU利用效率。这种技术本质上是为了将epoll作为定时器的核心驱动。

------

### 1. 为什么要动态调整超时时间？

要理解这个做法的必要性，我们需要先明白在一个事件循环中，定时器是如何被驱动的。你的程序（11-3）使用的是**独立的周期性信号（`SIGALRM`）** 来驱动定时检查，而清单11-4展示的是另一种更常见的模型：**将定时器与`epoll_wait`合并在同一个循环中**。

在这种合并模型中，`epoll_wait`的超时参数`timeout`扮演着双重角色：

- **有事件时**：它负责在指定时间内等待I/O事件，一旦有事件就立即返回处理。
- **无事件时**：它充当一个“定时器”，保证程序不会永久阻塞，而是在设定的超时时间到达后返回，以便我们可以去检查并处理那些到期的定时任务。

现在，我们来看一个典型的事件循环，它需要在每次循环中计算出**下一次应该等待多久**。

#### 场景分析：两种唤醒情况

假设我们有一个定时任务，需要在 `T_abs = 当前时间 + 100ms` 这个绝对时间点执行。

1. **情况A：被“定时器”唤醒**
   - 我们设置 `epoll_wait` 的超时参数为 `timeout = 100ms`。
   - 在这100ms内，没有任何I/O事件发生。`epoll_wait` 在精确的100ms后返回（返回值为0）。
   - **此时处理**：绝对时间 `T_abs` 已到，我们立刻执行定时任务。
   - **下一步**：设定下一个定时任务，并重新计算 `epoll_wait` 的 `timeout` 值。
2. **情况B：被“I/O事件”提前唤醒**
   - 我们同样设置 `epoll_wait` 的超时参数为 `timeout = 100ms`。
   - 但是，在第30ms时，一个客户端发来了数据。`epoll_wait` 被I/O事件提前唤醒，返回处理事件。
   - **此时问题**：虽然被唤醒了，但我们的定时任务（需要在100ms后执行）**还没有到期**。我们不能执行它。
   - **下一步**：处理完I/O事件后，我们需要再次调用 `epoll_wait` 来等待剩余的时间（`100ms - 30ms = 70ms`），直到定时任务真正到期 或 再次被“I/O事件”提前唤醒。**这时，就需要动态调整 `timeout` 参数**



# alarm函数

`alarm` 函数是一个简单的定时器接口，它允许进程在指定的**秒数**后收到一个 `SIGALRM` 信号。其原型如下：

```c
#include <unistd.h>
unsigned int alarm(unsigned int seconds);
```

### 使用方法

1. **设置定时器**：调用 `alarm(seconds)` 后，系统将在 `seconds` 秒后向当前进程发送 `SIGALRM` 信号。
2. **处理信号**：需要提前用 `signal` 或 `sigaction` 注册 `SIGALRM` 的处理函数，否则默认动作是终止进程。
3. **取消定时器**：调用 `alarm(0)` 可取消之前设置的闹钟，并返回之前剩余的时间。

### 返回值

- 返回之前设置的闹钟**剩余秒数**（如果有），否则返回 `0`。
- 如果本次调用覆盖了之前的闹钟，返回值可以帮助你决定是否需要重新安排任务。

### 注意事项

- **单一定时器**：每个进程只能有一个 `alarm` 定时器。再次调用 `alarm` 会重置剩余时间。
- **信号处理**：`SIGALRM` 默认会终止进程，所以必须捕获它。
- **精度**：以秒为单位，适合对时间精度要求不高的场景（如每隔几秒触发一次心跳检查）。
- **与 `sleep` 的区别**：`sleep` 会使进程阻塞指定时间，而 `alarm` 是异步的，程序可以继续执行其他任务，到期时通过信号通知。
- **与 `setitimer` 或 `timer_create` 对比**：`alarm` 功能简单，精度低；`setitimer` 支持微秒级和多种计时模式；`timer_create` 支持高精度和多种时钟源。

# `epoll_wait` 的`errno`错误码

在 Linux 系统编程中，`epoll_wait` 是一个系统调用，它的返回值有以下含义：

- **返回值 > 0**：表示有 `number` 个文件描述符上有事件发生（即就绪事件的数量）。
- **返回值 = 0**：表示在指定的超时时间内没有任何事件发生。（`timeout`参数表示 超时时间：-1阻塞，0立即返回，>0毫秒数；）
- **返回值 = -1**：表示调用失败，并设置 `errno` 来指示错误类型。

当 `epoll_wait` 返回 -1 时，我们需要根据 `errno` 的值判断错误的严重性：

- **`EINTR`**：系统调用被信号中断。这通常不是程序逻辑错误，而是一个临时情况。例如，当进程在执行 `epoll_wait` 期间收到了一个信号，并且该信号没有被设置为 `SA_RESTART` 标志时，`epoll_wait` 会提前返回 -1 并设置 `errno` 为 `EINTR`。这种情况下，程序通常可以选择**忽略这个错误，继续下一次 `epoll_wait` 调用**，因为被中断并不影响 epoll 实例本身的状态。

- **其他 `errno` 值**（如 `EBADF`、`EFAULT`、`EINVAL` 等）：表示真正的致命错误，比如 epoll 文件描述符无效、传入的事件数组指针有问题等。遇到这些错误时，程序一般无法继续正常运行，应该记录错误并退出循环。

因此，代码中判断：
```cpp
if((number < 0) && (errno != EINTR))
{
    printf("epoll failure\n");
    break;
}
```
正是为了区分这两种情况：
- 如果 `number < 0` 且 `errno != EINTR`，说明遇到了无法恢复的错误，于是打印信息并跳出循环（终止服务器）。
- 如果 `number < 0` 且 `errno == EINTR`，则忽略本次中断，继续下一次 `epoll_wait`（循环会再次执行，不会进入这个 if 分支）。

这种处理方式是网络编程中处理被信号中断的标准做法，能够保证程序的健壮性。

# 11.4 高性能定时器

## 11.4.1 时间轮

> 代码清单11-5 时间轮

执行定时器的核心代码位于 `tick()` 函数中：

```c
void time_wheel::tick() {
    tw_timer* tmp = slots[cur_slot];
    while (tmp) {
        if (tmp->rotation > 0) {
            tmp->rotation--;          // 圈数减1
        } else {
            tmp->cb_func(tmp->user_data);  // 执行回调-定时器任务。
            // 从链表中删除节点（略）
        }
        tmp = tmp->next;
    }
    cur_slot = (cur_slot + 1) % N;
}
```

### 执行效率

- **添加定时器**：O(1)
  - 计算圈数（`rotation`）和槽位（`ts`）只需除法和取模，与定时器总数无关。
  - 采用头插法将节点插入对应槽的链表，常数时间。
- **删除定时器**：O(1)
  - 已知节点指针，利用双向链表直接移除，前后指针调整均为常数操作。
- **执行一个定时器**：O(1)（指单个定时器从被扫描到回调执行完毕）
  - 判断圈数、调用回调、删除节点均不依赖其他定时器数量。
- **`tick()` 函数**每时间间隔（`SI`）处理当前指针指向的整个槽位链表。
  - 该过程的时间复杂度为 **O(当前槽定时器数量)**，而非 O(总定时器数)。
  - 若所有定时器集中在少数槽，这些槽的 `tick()` 会变慢；但通过增加槽数可有效降低平均链表长度，使整体性能可控。

###  多级时间轮的优势

- 单级时间轮通过增加槽数可缩短链表，但槽数受内存限制。
- 多级时间轮（如 Linux 内核定时器）将定时器按超时范围分层，每层槽数固定。
  - 添加定时器时根据超时时间分配到对应层级，每个层级链表短。
  - `tick()` 处理时仅操作当前层的一个槽，当该层转完一圈，将下一层的一个槽的定时器降级到更细层级。
  - 每个定时器的处理仅涉及常数次操作，整体接近 **O(1)**。
